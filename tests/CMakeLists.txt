add_executable(nvrhi_memory_queries memory-queries.cpp)
target_compile_definitions(nvrhi_memory_queries PRIVATE TEST_SHARED=$<BOOL:${NVRHI_BUILD_SHARED}>)
if(NVRHI_WITH_DX11)
    target_link_libraries(nvrhi_memory_queries PRIVATE ${nvrhi_d3d11_target})
    target_compile_definitions(nvrhi_memory_queries PRIVATE TEST_D3D11=1)
    add_test(NAME memory_queries_d3d11 COMMAND nvrhi_memory_queries d3d11)
endif()
if(NVRHI_WITH_DX12)
    target_link_libraries(nvrhi_memory_queries PRIVATE ${nvrhi_d3d12_target})
    target_compile_definitions(nvrhi_memory_queries PRIVATE TEST_D3D12=1)
    add_test(NAME memory_queries_d3d12 COMMAND nvrhi_memory_queries d3d12)
endif()
if(NVRHI_WITH_VULKAN)
    target_link_libraries(nvrhi_memory_queries PRIVATE ${nvrhi_vulkan_target}
        ${nvrhi_vulkan_headers_target} ${CMAKE_DL_LIBS})
    target_compile_definitions(nvrhi_memory_queries PRIVATE TEST_VULKAN=1)
    add_test(NAME memory_queries_vulkan COMMAND nvrhi_memory_queries vulkan)
endif()
if(NVRHI_WITH_VALIDATION)
    target_compile_definitions(nvrhi_memory_queries PRIVATE TEST_VALIDATION=1)
endif()
target_link_libraries(nvrhi_memory_queries PRIVATE nvrhi)
get_property(memory_tests DIRECTORY PROPERTY TESTS)
set_tests_properties(${memory_tests} PROPERTIES TIMEOUT 30)
